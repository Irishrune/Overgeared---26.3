#!/usr/bin/env python3
"""Conservative class-name migrations confirmed against Minecraft 26.3 mappings.

This is NOT an automatic API port. Constructors and methods differ radically.
Only import-qualified files are edited, and symbol renames occur within those
files to keep their references internally consistent. Repeatable/idempotent.
"""
from __future__ import annotations
import argparse
import re
from collections import Counter
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
JAVA_ROOTS = [ROOT / 'common/src/main/java', ROOT / 'fabric/src/main/java']
MAPPINGS = {
    # Verified via mappings.relativitymc.org/26.3 Minecraft class descriptors.
    'net.minecraft.world.level.Level': 'net.minecraft.world.World',
    'net.minecraft.core.BlockPos': 'net.minecraft.util.math.BlockPos',
    'net.minecraft.core.Direction': 'net.minecraft.util.math.Direction',
    'net.minecraft.network.chat.Component': 'net.minecraft.text.Text',
    'net.minecraft.world.entity.player.Player': 'net.minecraft.entity.player.PlayerEntity',
    'net.minecraft.world.entity.player.Inventory': 'net.minecraft.entity.player.PlayerInventory',
    'net.minecraft.world.item.crafting.Ingredient': 'net.minecraft.recipe.Ingredient',
    'net.minecraft.world.item.crafting.RecipeHolder': 'net.minecraft.recipe.RecipeEntry',
    'net.minecraft.world.item.crafting.RecipeSerializer': 'net.minecraft.recipe.RecipeSerializer',
    'net.minecraft.world.item.crafting.RecipeType': 'net.minecraft.recipe.RecipeType',
    'net.minecraft.network.codec.StreamCodec': 'net.minecraft.network.codec.PacketCodec',
    'net.minecraft.network.codec.ByteBufCodecs': 'net.minecraft.network.codec.PacketCodecs',
    'net.minecraft.network.RegistryFriendlyByteBuf': 'net.minecraft.network.RegistryByteBuf',
    'net.minecraft.core.NonNullList': 'net.minecraft.util.collection.DefaultedList',
}


def port_text(data: str) -> tuple[str, list[str]]:
    # Fully qualified uses are replaced independent of imports, but only when
    # they exist in Java (not Markdown/data packs). Do them first, in decreasing
    # source length order to prevent accidental prefix overlap.
    changed = []
    for old, new in sorted(MAPPINGS.items(), key=lambda x: -len(x[0])):
        if old not in data:
            continue
        old_name, new_name = old.rsplit('.', 1)[-1], new.rsplit('.', 1)[-1]
        old_import = f'import {old};'
        had_import = old_import in data
        has_fqn = old in data
        data = re.sub(r'(?<![\w.$])' + re.escape(old) + r'(?![\w$])', new, data)
        # The imported symbol now has a new simple class name. Rewriting all
        # references is safe only when the old class was explicitly imported;
        # otherwise another type of the same simple name may be in use.
        if had_import and old_name != new_name:
            data = re.sub(r'\b' + re.escape(old_name) + r'\b', new_name, data)
        if had_import or has_fqn:
            changed.append(old)
    return data, changed


def run(check: bool = False) -> tuple[int, Counter]:
    files_changed = 0
    symbol_count = Counter()
    for java_root in JAVA_ROOTS:
        for java_file in sorted(java_root.rglob('*.java')):
            original = java_file.read_text(encoding='utf-8')
            migrated, changes = port_text(original)
            if migrated != original:
                files_changed += 1
                symbol_count.update(changes)
                if not check:
                    java_file.write_text(migrated, encoding='utf-8')
    return files_changed, symbol_count


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--check', action='store_true', help='require no pending class-name migrations')
    args = parser.parse_args()
    count, symbols = run(args.check)
    print(f'Mapped class names: {sum(symbols.values())} file-symbol migrations in {count} Java files')
    for symbol, n in symbols.most_common():
        print(f'  {n:3} {symbol} => {MAPPINGS[symbol]}')
    if args.check and count:
        raise SystemExit(1)


if __name__ == '__main__':
    main()
